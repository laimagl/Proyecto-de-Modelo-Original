def make_dataloader(dataset, batch_size=64, train=True):
    """
    Returns a DataLoader tuned for Colab's CPU count and GPU transfer.
    pin_memory   → faster CPU→GPU transfer via pinned (page-locked) memory
    persistent_workers → workers stay alive between epochs (saves fork time)
    prefetch_factor  → each worker pre-fetches this many batches ahead
    """
    return torch.utils.data.DataLoader(
        dataset,
        batch_size      = batch_size,
        shuffle         = train,
        num_workers     = min(4, cpu_cores),   # Colab caps at 4 usable workers
        pin_memory      = torch.cuda.is_available(),
        persistent_workers = True,
        prefetch_factor = 2,
        drop_last       = train,               # avoid small last batch issues
    )


# ── 7. GRADIENT ACCUMULATION TEMPLATE ────────────────────────
# Simulates a larger batch without extra VRAM.
# effective_batch = batch_size × accumulation_steps
ACCUMULATION_STEPS = 4   # ← tune this

def train_step_with_accumulation(model, optimizer, dataloader, loss_fn):
    model.train()
    optimizer.zero_grad()
    for step, (X, y) in enumerate(dataloader):
        X, y = X.cuda(), y.cuda()
        with autocast():
            pred = model(X)
            loss = loss_fn(pred, y) / ACCUMULATION_STEPS   # normalize
        scaler.scale(loss).backward()
        if (step + 1) % ACCUMULATION_STEPS == 0:
            scaler.step(optimizer)
            scaler.update()
            optimizer.zero_grad()

    path = f"{DRIVE_PATH}/{filename}"
    if not os.path.exists(path):
        print("  No checkpoint found, starting fresh.")
        return 0, None
    ckpt = torch.load(path, map_location="cuda" if torch.cuda.is_available() else "cpu")
    model.load_state_dict(ckpt["model"])
    optimizer.load_state_dict(ckpt["optimizer"])
    print(f"  Checkpoint loaded (epoch {ckpt['epoch']})")
    return ckpt["epoch"], ckpt["loss"]

def benchmark_gpu(size=4096, reps=20):
    """Matmul throughput test — useful to compare T4 vs A100."""
    if not torch.cuda.is_available():
        print("  Benchmark skipped — no GPU"); return
    A = torch.randn(size, size, device="cuda", dtype=torch.float16)
    B = torch.randn(size, size, device="cuda", dtype=torch.float16)
    torch.cuda.synchronize()
    t0 = time.perf_counter()
    for _ in range(reps):
        torch.matmul(A, B)
    torch.cuda.synchronize()
    elapsed = time.perf_counter() - t0
    tflops = (2 * size**3 * reps) / elapsed / 1e12
    print(f"  Benchmark: {tflops:.1f} TFLOPS  ({elapsed*1000/reps:.1f} ms/iter) — {torch.cuda.get_device_name(0)}")

def firstsmooth(y, lambda_, start=None):
    ytilde = y.copy()
    if start is None:
        start = y[0]
    ytilde[0] = lambda_ * y[0] + (1 - lambda_) * start
    for i in range(1, len(y)):
        ytilde[i] = lambda_ * y[i] + (1 - lambda_) * ytilde[i - 1]
    return ytilde

def measacc_fs(y, lambda_):
    out = firstsmooth(y, lambda_)
    T = len(y)
    yh = y.copy().values
    out = pd.concat([pd.Series([y[0]]), out.iloc[:-1]], ignore_index=True).values
    prederr = yh - out
    SSE = sum(prederr**2)
    RMSE = np.sqrt(sum(prederr**2) / T) # Usamos RMSE dado que en este tipo de series, lo valores cercanos a 0 son comunes
    MAD = sum(abs(prederr)) / T
    MSD = sum(prederr**2) / T
    ret1 = pd.DataFrame({
        "SSE": [SSE],
        "RMSE": [RMSE],
        "MAD": [MAD],
        "MSD": [MSD]
    })
    ret1.reset_index(drop=True, inplace=True)
    return ret1

def measacc_hw(y, lambda_):
    model = ExponentialSmoothing(y.values)
    fit = model.fit(smoothing_level=lambda_)
    T = len(y)
    yh = y.copy().values
    y_pred = pd.Series(data=fit.fittedvalues, index=y.index)
    out = pd.concat([pd.Series([y[0]]), y_pred[:-1]], ignore_index=True).values
    prederr = yh - out
    SSE = sum(prederr**2)
    RMSE = np.sqrt(sum(prederr**2) / T)
    MAD = sum(abs(prederr)) / T
    MSD = sum(prederr**2) / T
    ret1 = pd.DataFrame({
        "SSE": [SSE],
        "RMSE": [RMSE],
        "MAD": [MAD],
        "MSD": [MSD]
    })
    ret1.reset_index(drop=True, inplace=True)
    return ret1

El suavizado exponencial simple resulta ser el mejor modelo en este caso porque los retornos del EUR/USD no presentan tendencia ni estacionalidad marcada y se mantienen alrededor de una media cercana a cero. Al no existir una estructura clara en la serie, métodos más complejos como Holt-Winters, que incorporan tendencia y estacionalidad, terminan introduciendo más ruido que información útil. Por ello, en series financieras como los retornos de divisas, donde predomina el comportamiento aleatorio, los modelos más simples y parsimoniosos suelen ofrecer mejores resultados.

def tlsmooth(y, delta_, y_tilde_start=None, lambda_start=1):
    T = len(y)

    Qt = np.zeros(T)
    Dt = np.zeros(T)
    y_tilde = np.zeros(T)
    lambd = np.zeros(T)
    err = np.zeros(T)

    lambd[0] = lambda_start
    if y_tilde_start is None:
        y_tilde[0] = y[0]
    else:
        y_tilde[0] = y_tilde_start

    for i in range(1, T):
        err[i] = y[i] - y_tilde[i-1]
        Qt[i] = delta_ * err[i] + (1 - delta_) * Qt[i-1]
        Dt[i] = delta_ * abs(err[i]) + (1 - delta_) * Dt[i-1]
        lambd[i] = abs(Qt[i] / Dt[i])
        y_tilde[i] = lambd[i] * y[i] + (1 - lambd[i]) * y_tilde[i-1]

    return np.column_stack((y_tilde, lambd, err, Qt, Dt))

def tes_model_tuning(train, test, step, trend, seasonal, seasonal_periods,
                     title="Model Tuning - Triple Exponential Smoothing"):

    train = np.asarray(train, dtype=float)
    test = np.asarray(test, dtype=float)

    # 🔥 chequeo clave
    if np.isnan(train).any() or np.isnan(test).any():
        raise ValueError("Hay NaN en train o test. Limpia los datos primero.")

    alphas = betas = gammas = np.arange(0.10, 1, 0.10)
    abg = list(itertools.product(alphas, betas, gammas))

    best_alpha, best_beta, best_gamma, best_mae = tes_optimizer(
        train, test, abg,
        trend=trend,
        seasonal=seasonal,
        seasonal_periods=seasonal_periods,
        step=step
    )

        seasonal_periods=seasonal_periods

        smoothing_trend=best_beta,   # 🔥 CORREGIDO
        smoothing_seasonal=best_gamma,
        optimized=False
    )

    y_pred = final_model.forecast(step)

    mae = mean_absolute_error(test[:step], y_pred)

    plot_model(train, test, y_pred, title)
