import os
import gc
import itertools
import numpy as np
import pandas as pd
import tensorflow as tf
import kagglehub
import matplotlib.pyplot as plt

from sklearn.metrics import mean_squared_error, mean_absolute_error, mean_absolute_percentage_error
from sklearn.preprocessing import StandardScaler
from statsmodels.stats.diagnostic import acorr_ljungbox

# =========================
# 0. Configuración
# =========================
SEED = 42
np.random.seed(SEED)
tf.random.set_seed(SEED)

try:
    gpus = tf.config.list_physical_devices('GPU')
    if gpus:
        for gpu in gpus:
            tf.config.experimental.set_memory_growth(gpu, True)
except Exception:
    pass

# =========================
# 1. Datos
# =========================
path = kagglehub.dataset_download("imetomi/eur-usd-forex-pair-historical-data-2002-2019")
df = pd.read_csv(os.path.join(path, "eurusd_minute.csv"))

df_time = df.copy()

if 'Date' in df_time.columns and 'Time' in df_time.columns:
    df_time['Datetime'] = pd.to_datetime(df_time['Date'] + ' ' + df_time['Time'])
    df_time = df_time.set_index('Datetime')
    df_time = df_time.drop(columns=['Date', 'Time'])
    df_time = df_time.sort_index()

# Retorno porcentual
df_time["Retorno"] = df_time["BC"].pct_change()
df_time = df_time.dropna()

# Tomamos solo 1000 datos para aligerar
df_time = df_time.tail(10000).copy()

serie = df_time["Retorno"].values.astype(np.float32)
fechas = df_time.index

print(f"Número de observaciones usadas: {len(serie)}")

# =========================
# 2. Función para crear ventanas
# =========================
def crear_dataset(serie, window_size=7):
    X, y = [], []
    for i in range(len(serie) - window_size):
        X.append(serie[i:i+window_size])
        y.append(serie[i+window_size])
    return np.array(X), np.array(y)

window_size = 7
X, y = crear_dataset(serie, window_size=window_size)

print("Shape X:", X.shape)
print("Shape y:", y.shape)

# =========================
# 3. Split final train+val / test
# =========================
# Dejamos el último 15% para test final fuera de muestra
test_size = int(len(X) * 0.15)

X_trainval = X[:-test_size]
y_trainval = y[:-test_size]

X_test = X[-test_size:]
y_test = y[-test_size:]

print("Train+Val:", X_trainval.shape, y_trainval.shape)
print("Test:", X_test.shape, y_test.shape)

# =========================
# 4. Función rolling / walk-forward splits
# =========================
def rolling_splits(X, y, train_ratio=0.6, val_ratio=0.2, n_splits=3):
    """
    Genera splits temporales tipo walk-forward dentro de train+val.
    """
    n = len(X)
    train_size = int(n * train_ratio)
    val_size = int(n * val_ratio)

    splits = []
    start_train = 0

    for i in range(n_splits):
        train_end = train_size + i * val_size // 2
        val_end = train_end + val_size

        if val_end > n:
            break

        X_tr = X[start_train:train_end]
        y_tr = y[start_train:train_end]
        X_val = X[train_end:val_end]
        y_val = y[train_end:val_end]

        splits.append((X_tr, y_tr, X_val, y_val))

    return splits

splits = rolling_splits(X_trainval, y_trainval, train_ratio=0.6, val_ratio=0.2, n_splits=3)
print(f"Número de rolling splits: {len(splits)}")

# =========================
# 5. Escalado
# =========================
def escalar_datasets(X_tr, X_val, X_te=None):
    scaler = StandardScaler()

    X_tr_2d = X_tr.reshape(-1, 1)
    scaler.fit(X_tr_2d)

    X_tr_scaled = scaler.transform(X_tr.reshape(-1, 1)).reshape(X_tr.shape)
    X_val_scaled = scaler.transform(X_val.reshape(-1, 1)).reshape(X_val.shape)

    if X_te is not None:
        X_te_scaled = scaler.transform(X_te.reshape(-1, 1)).reshape(X_te.shape)
        return X_tr_scaled, X_val_scaled, X_te_scaled, scaler

    return X_tr_scaled, X_val_scaled, scaler

# =========================
# 6. Modelo MLP
# =========================
def construir_mlp(input_dim, units=(64, 32), dropout_rate=0.2, learning_rate=0.001):
    model = tf.keras.Sequential()
    model.add(tf.keras.layers.Input(shape=(input_dim,)))

    for u in units:
        model.add(tf.keras.layers.Dense(u, activation="tanh"))
        if dropout_rate > 0:
            model.add(tf.keras.layers.Dropout(dropout_rate))

    model.add(tf.keras.layers.Dense(1))

    optimizer = tf.keras.optimizers.Adam(learning_rate=learning_rate)

    model.compile(
        optimizer=optimizer,
        loss="mse",
        metrics=[tf.keras.metrics.RootMeanSquaredError()]
    )
    return model

# =========================
# 7. Grid de hiperparámetros (ligero)
# =========================
param_grid = {
    "units": [(32,), (64,), (64, 32), (128, 64)],
    "dropout_rate": [0.0, 0.2],
    "learning_rate": [0.001, 0.0005],
    "batch_size": [16, 32]
}

combinaciones = list(itertools.product(
    param_grid["units"],
    param_grid["dropout_rate"],
    param_grid["learning_rate"],
    param_grid["batch_size"]
))

print(f"Total combinaciones a probar: {len(combinaciones)}")

# =========================
# 8. Entrenamiento con rolling validation
# =========================
resultados = []

for units, dropout_rate, learning_rate, batch_size in combinaciones:
    print(f"\nProbando units={units}, dropout={dropout_rate}, lr={learning_rate}, batch={batch_size}")

    rmses_val = []
    histories = []

    for fold, (X_tr, y_tr, X_val, y_val) in enumerate(splits, start=1):
        tf.keras.backend.clear_session()
        gc.collect()

        X_tr_sc, X_val_sc, scaler = escalar_datasets(X_tr, X_val)

        model = construir_mlp(
            input_dim=window_size,
            units=units,
            dropout_rate=dropout_rate,
            learning_rate=learning_rate
        )

        early_stop = tf.keras.callbacks.EarlyStopping(
            monitor="val_loss",
            patience=10,
            restore_best_weights=True,
            verbose=0
        )

        reduce_lr = tf.keras.callbacks.ReduceLROnPlateau(
            monitor="val_loss",
            factor=0.5,
            patience=5,
            min_lr=1e-5,
            verbose=0
        )

        history = model.fit(
            X_tr_sc, y_tr,
            validation_data=(X_val_sc, y_val),
            epochs=100,
            batch_size=batch_size,
            callbacks=[early_stop, reduce_lr],
            verbose=0
        )

        y_val_pred = model.predict(X_val_sc, verbose=0).flatten()
        rmse_val = np.sqrt(mean_squared_error(y_val, y_val_pred))

        rmses_val.append(rmse_val)
        histories.append(history.history)

        print(f"  Fold {fold} - RMSE val: {rmse_val:.8f}")

    resultados.append({
        "units": units,
        "dropout_rate": dropout_rate,
        "learning_rate": learning_rate,
        "batch_size": batch_size,
        "mean_val_rmse": np.mean(rmses_val),
        "std_val_rmse": np.std(rmses_val),
        "histories": histories
    })

# =========================
# 9. Ranking
# =========================
resultados_df = pd.DataFrame(resultados).sort_values(
    by=["mean_val_rmse", "std_val_rmse"],
    ascending=[True, True]
).reset_index(drop=True)

print("\nTop 10 configuraciones:")
print(resultados_df.head(10)[[
    "units", "dropout_rate", "learning_rate", "batch_size",
    "mean_val_rmse", "std_val_rmse"
]])

best_config = resultados_df.iloc[0]
print("\nMejor configuración encontrada:")
print(best_config[[
    "units", "dropout_rate", "learning_rate", "batch_size",
    "mean_val_rmse", "std_val_rmse"
]])

# =========================
# 10. Reentrenar mejor modelo en train+val
# =========================
tf.keras.backend.clear_session()
gc.collect()

# Para el entrenamiento final, separamos una pequeña validación temporal interna
val_final_size = int(len(X_trainval) * 0.15)

X_train_final = X_trainval[:-val_final_size]
y_train_final = y_trainval[:-val_final_size]

X_val_final = X_trainval[-val_final_size:]
y_val_final = y_trainval[-val_final_size:]

X_train_final_sc, X_val_final_sc, X_test_sc, scaler_final = escalar_datasets(
    X_train_final, X_val_final, X_test
)

best_model = construir_mlp(
    input_dim=window_size,
    units=best_config["units"],
    dropout_rate=best_config["dropout_rate"],
    learning_rate=best_config["learning_rate"]
)

early_stop = tf.keras.callbacks.EarlyStopping(
    monitor="val_loss",
    patience=12,
    restore_best_weights=True,
    verbose=1
)

reduce_lr = tf.keras.callbacks.ReduceLROnPlateau(
    monitor="val_loss",
    factor=0.5,
    patience=6,
    min_lr=1e-5,
    verbose=1
)

history_final = best_model.fit(
    X_train_final_sc, y_train_final,
    validation_data=(X_val_final_sc, y_val_final),
    epochs=150,
    batch_size=int(best_config["batch_size"]),
    callbacks=[early_stop, reduce_lr],
    verbose=1
)

# =========================
# 11. Evaluación en test
# =========================
y_test_pred = best_model.predict(X_test_sc, verbose=0).flatten()

rmse_test = np.sqrt(mean_squared_error(y_test, y_test_pred))
mae_test = mean_absolute_error(y_test, y_test_pred)
mape_test = mean_absolute_percentage_error(y_test + 1e-8, y_test_pred)

residuos = y_test - y_test_pred

try:
    lb = acorr_ljungbox(residuos, lags=[10], return_df=True)
    lb_pvalue = lb["lb_pvalue"].iloc[0]
except Exception:
    lb_pvalue = np.nan

# Accuracy direccional
dir_real = np.sign(y_test)
dir_pred = np.sign(y_test_pred)
directional_accuracy = np.mean(dir_real == dir_pred)

print("\n===== RESULTADOS FINALES =====")
print(f"RMSE test: {rmse_test:.8f}")
print(f"MAE test:  {mae_test:.8f}")
print(f"MAPE test: {mape_test:.8f}")
print(f"Ljung-Box p-value (lag 10): {lb_pvalue:.6f}")
print(f"Directional Accuracy: {directional_accuracy:.4f}")

# =========================
# 12. Gráficos
# =========================
plt.figure(figsize=(10, 4))
plt.plot(history_final.history["root_mean_squared_error"], label="Train RMSE")
plt.plot(history_final.history["val_root_mean_squared_error"], label="Val RMSE")
plt.title("Curvas de aprendizaje - Mejor modelo")
plt.xlabel("Épocas")
plt.ylabel("RMSE")
plt.legend()
plt.grid(True)
plt.show()

plt.figure(figsize=(12, 5))
plt.plot(y_test[:200], label="Real")
plt.plot(y_test_pred[:200], label="Predicción")
plt.title("Predicción vs Real (primeras 200 observaciones del test)")
plt.xlabel("Tiempo")
plt.ylabel("Retorno")
plt.legend()
plt.grid(True)
plt.show()

plt.figure(figsize=(12, 4))
plt.plot(residuos, label="Residuos")
plt.title("Residuos del mejor modelo")
plt.xlabel("Tiempo")
plt.ylabel("Error")
plt.legend()
plt.grid(True)
plt.show()

def construir_rnn(input_shape, units=(32,), dropout_rate=0.2, learning_rate=0.001):

    for i, u in enumerate(units):
        return_sequences = i < len(units) - 1
        model.add(

                activation="tanh",
                return_sequences=return_sequences

        )

    model.add(tf.keras.layers.Dense(1))

    optimizer = tf.keras.optimizers.Adam(learning_rate=learning_rate)

    model.compile(
        optimizer=optimizer,
        loss="mse",
        metrics=[tf.keras.metrics.RootMeanSquaredError()]
    )
    return model

# 4. Función rolling / walk-forward splits
# =========================
def rolling_splits(X, y, train_ratio=0.6, val_ratio=0.2, n_splits=3):
    """
    Genera splits temporales tipo walk-forward dentro de train+val.
    """
    n = len(X)
    train_size = int(n * train_ratio)
    val_size = int(n * val_ratio)

    splits = []
    start_train = 0

    for i in range(n_splits):
        train_end = train_size + i * val_size // 2
        val_end = train_end + val_size

        if val_end > n:
            break

        X_tr = X[start_train:train_end]
        y_tr = y[start_train:train_end]
        X_val = X[train_end:val_end]
        y_val = y[train_end:val_end]

        splits.append((X_tr, y_tr, X_val, y_val))

    return splits

splits = rolling_splits(X_trainval, y_trainval, train_ratio=0.6, val_ratio=0.2, n_splits=3)

def escalar_datasets_lstm(X_tr, X_val, X_te=None):
    scaler = StandardScaler()

    # Ajustar scaler usando solo train
    X_tr_2d = X_tr.reshape(-1, 1)
    scaler.fit(X_tr_2d)

    X_tr_scaled = scaler.transform(X_tr.reshape(-1, 1)).reshape(X_tr.shape)
    X_val_scaled = scaler.transform(X_val.reshape(-1, 1)).reshape(X_val.shape)

    if X_te is not None:
        X_te_scaled = scaler.transform(X_te.reshape(-1, 1)).reshape(X_te.shape)
        return X_tr_scaled, X_val_scaled, X_te_scaled, scaler

    return X_tr_scaled, X_val_scaled, scaler

# =========================
# 6. Modelo LSTM
# =========================
def construir_lstm(input_shape, units=(32,), dropout_rate=0.2, learning_rate=0.001):
    model = tf.keras.Sequential()
    model.add(tf.keras.layers.Input(shape=input_shape))

    for i, u in enumerate(units):
        return_sequences = i < len(units) - 1
        model.add(
            tf.keras.layers.LSTM(
                u,
                activation="tanh",
                recurrent_activation="sigmoid",
                return_sequences=return_sequences
            )
        )
        if dropout_rate > 0:
            model.add(tf.keras.layers.Dropout(dropout_rate))

    model.add(tf.keras.layers.Dense(1))

    optimizer = tf.keras.optimizers.Adam(learning_rate=learning_rate)

    model.compile(
        optimizer=optimizer,
        loss="mse",
        metrics=[tf.keras.metrics.RootMeanSquaredError()]
    )
    return model
