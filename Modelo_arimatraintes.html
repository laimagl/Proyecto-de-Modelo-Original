import torch

# ── 1. Verificar GPU ─────────────────────────────────────────
assert torch.cuda.is_available(), "No hay GPU. Cambia el runtime a L4."
device = torch.device('cuda')

print(f"GPU detectada : {gpu_name}")
print(f"VRAM total    : {vram_gb:.1f} GB")

# ── 2. Optimizaciones del sistema ────────────────────────────
torch.backends.cuda.matmul.allow_tf32 = True   # TF32 en matmul → más rápido

print("Optimizaciones del sistema: OK")

# ── 3. Configurar streams CUDA ───────────────────────────────
#  Permite solapar cómputo y transferencias CPU↔GPU
compute_stream  = torch.cuda.Stream()
transfer_stream = torch.cuda.Stream()
print("CUDA streams: OK")

# ── 4. Helpers de memoria ────────────────────────────────────
def vram():
    used  = torch.cuda.memory_allocated() / 1e9
    total = torch.cuda.get_device_properties(0).total_memory / 1e9
    print(f"VRAM usada: {used:.2f} / {total:.1f} GB ({used/total*100:.1f}%)")

def clear_gpu():
    torch.cuda.empty_cache()
    torch.cuda.ipc_collect()
    print("Cache GPU limpiada")

# ── 5. Entrenamiento con AMP (Automatic Mixed Precision) ─────
#  AMP usa float16 donde puede → el doble de throughput en la L4
def train_con_amp(model, loader, epochs=3, lr=1e-3):
    model = model.to(device)
    optimizer = torch.optim.AdamW(model.parameters(), lr=lr, fused=True)  # fused → más rápido en CUDA
    criterion = nn.CrossEntropyLoss()
    scaler    = torch.cuda.amp.GradScaler()  # escala pérdidas para FP16

    for epoch in range(epochs):
        model.train()
        total_loss, t0 = 0, time.time()

        for xb, yb in loader:
            xb = xb.to(device, non_blocking=True)  # non_blocking → solapado con cómputo
            yb = yb.to(device, non_blocking=True)

            optimizer.zero_grad(set_to_none=True)  # más rápido que zero_grad()

            with torch.autocast(device_type='cuda', dtype=torch.float16):
                loss = criterion(model(xb), yb)

            scaler.scale(loss).backward()
            scaler.unscale_(optimizer)
            nn.utils.clip_grad_norm_(model.parameters(), 1.0)  # estabilidad
            scaler.step(optimizer)
            scaler.update()

            total_loss += loss.item()

        elapsed = time.time() - t0
        print(f"Epoch {epoch+1}/{epochs} | Loss: {total_loss/len(loader):.4f} | {elapsed:.1f}s")

    return model

# ── 6. Demo: modelo + datos de prueba ────────────────────────
model = nn.Sequential(
    nn.Linear(256, 1024), nn.GELU(),
    nn.Linear(1024, 512),  nn.GELU(),
    nn.Linear(512, 10)
)

# pin_memory=True → transfiere datos a GPU mucho más rápido
from torch.utils.data import DataLoader, TensorDataset
X  = torch.randn(20_000, 256)
y  = torch.randint(0, 10, (20_000,))
ds = TensorDataset(X, y)
loader = DataLoader(
    ds,
    batch_size  = 1024,
    shuffle     = True,
    num_workers = 2,      # carga datos en paralelo
    pin_memory  = True,   # DMA directo a GPU
    prefetch_factor = 2   # prefetch de batches
)

print("\nIniciando entrenamiento optimizado...")
vram()
model = train_con_amp(model, loader)
print("\nListo. Estado final de VRAM:")
vram()

    mape = np.mean(np.abs(forecast - actual)/np.abs(actual)) # MAPE
    mae = np.mean(np.abs(forecast - actual))                 # MAE

    mse = np.mean((forecast - actual)**2)                    # MSE

                           'MSE': [mse],

                           'RMSE': [rmse]},
